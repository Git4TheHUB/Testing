REM ================================================
REM OLD ROBLOX - TI-84 Plus CE (TI-BASIC)
REM Isometric 3D-style baseplate + player + jump
REM Inspired by TICraft-style blocky 3D effect
REM (Best pure TI-BASIC approximation possible)
REM ================================================
REM How to make .8xp:
REM 1. Copy this entire code into https://www.cemetech.net/sc/ (SourceCoder)
REM 2. Select TI-84 Plus CE, compile to .8xp
REM 3. Send to calculator with TI Connect CE
REM 4. Run from prgm menu (name will be OLDROBLOX or similar)
REM ================================================

ClrDraw
AxesOff
0→Xmin:1→Xmax
0→Ymin:1→Ymax
Text(0,0,"OLD ROBLOX")
Text(10,0,"Arrows:Move  ENTER:Jump  CLEAR:Exit")
Pause 

REM === VARIABLES ===
REM PX,PZ = player map position (isometric grid)
REM PY = player height (for jumping)
REM VX,VY = velocity for jump physics
REM CX,CY = cursor position (old Roblox arrow)
REM SCALE = isometric tile size
REM OX,OY = screen offset for isometric origin

10→SCALE
80→OX
60→OY
0→PX:0→PZ:0→PY
0→VX:0→VY
160→CX:100→CY
0→ONGROUND:1→ONGROUND

REM === DRAW ISOMETRIC BASEPLATE (TICraft-like grid) ===
Lbl DRAWBASE
ClrDraw
REM Draw ground plane as isometric grid (parallelograms/diamonds)
For(I,0,8)
For(J,0,8)
  REM Isometric projection: screenX = (I-J)*SCALE/2 + OX
  REM screenY = (I+J)*SCALE/4 + OY
  (I-J)*SCALE/2+OX→SX1
  (I+J)*SCALE/4+OY→SY1
  (I-J+1)*SCALE/2+OX→SX2
  (I+J+1)*SCALE/4+OY→SY2
  (I-J)*SCALE/2+OX→SX3
  (I+J+1)*SCALE/4+OY→SY3
  (I-J+1)*SCALE/2+OX→SX4
  (I+J)*SCALE/4+OY→SY4
  REM Draw tile edges (gray for baseplate)
  Line(SX1,SY1,SX2,SY2,0)  REM dark gray
  Line(SX2,SY2,SX3,SY3,0)
  Line(SX3,SY3,SX4,SY4,0)
  Line(SX4,SY4,SX1,SY1,0)
End
End

REM Draw some "bricks" or 3D cubes on the baseplate (for depth)
REM Example cube at (3,3)
3→I:3→J:1→H  REM H=height in blocks
(I-J)*SCALE/2+OX→SX
(I+J)*SCALE/4+OY→SY
REM Simple isometric cube (top + sides)
Line(SX,SY-H*SCALE/3,SX+SCALE/2,SY-SCALE/6-H*SCALE/3,0)
Line(SX+SCALE/2,SY-SCALE/6-H*SCALE/3,SX,SY-SCALE/3-H*SCALE/3,0)
Line(SX,SY-SCALE/3-H*SCALE/3,SX-SCALE/2,SY-SCALE/6-H*SCALE/3,0)
Line(SX-SCALE/2,SY-SCALE/6-H*SCALE/3,SX,SY-H*SCALE/3,0)
REM Vertical edges for 3D
Line(SX,SY,SX,SY-H*SCALE/3,0)
Line(SX+SCALE/2,SY-SCALE/6,SX+SCALE/2,SY-SCALE/6-H*SCALE/3,0)
Line(SX-SCALE/2,SY-SCALE/6,SX-SCALE/2,SY-SCALE/6-H*SCALE/3,0)

REM === DRAW PLAYER (blocky Roblox noob style) ===
Lbl DRAWPLAYER
REM Project player position
(PX-PZ)*SCALE/2+OX→PSX
(PX+PZ)*SCALE/4+OY-PY*SCALE/6→PSY   REM subtract PY for jump height

REM Body (blue block)
Line(PSX-4,PSY-12,PSX+4,PSY-12,0)
Line(PSX+4,PSY-12,PSX+4,PSY-4,0)
Line(PSX+4,PSY-4,PSX-4,PSY-4,0)
Line(PSX-4,PSY-4,PSX-4,PSY-12,0)

REM Head (yellow)
Line(PSX-3,PSY-18,PSX+3,PSY-18,0)
Line(PSX+3,PSY-18,PSX+3,PSY-12,0)
Line(PSX+3,PSY-12,PSX-3,PSY-12,0)
Line(PSX-3,PSY-12,PSX-3,PSY-18,0)

REM Legs (animated simple - static for TI-BASIC speed)
Line(PSX-3,PSY-4,PSX-3,PSY+2,0)
Line(PSX+3,PSY-4,PSX+3,PSY+2,0)

REM Arms
Line(PSX-4,PSY-10,PSX-8,PSY-6,0)
Line(PSX+4,PSY-10,PSX+8,PSY-6,0)

Return

REM === DRAW OLD ROBLOX CURSOR (arrow) ===
Lbl DRAWCURSOR
REM Simple arrow at (CX,CY)
Line(CX,CY,CX,CY+12,0)
Line(CX,CY,CX+8,CY+8,0)
Line(CX+1,CY+1,CX+1,CY+10,0)
Line(CX+8,CY+8,CX+3,CY+8,0)
Return

REM === MAIN GAME LOOP ===
Lbl LOOP
getKey→K

REM Movement (arrow keys: 24=left, 25=right, 26=up, 34=down)
If K=24: PX-1→PX
If K=25: PX+1→PX
If K=26: PZ-1→PZ
If K=34: PZ+1→PZ

REM Jump (ENTER = 105)
If K=105 and ONGROUND=1
Then
  6→VY
  0→ONGROUND
End

REM Physics
VY-1→VY
PY+VY→PY
If PY<0
Then
  0→PY
  0→VY
  1→ONGROUND
End

REM Clamp to baseplate
If PX<0:0→PX
If PX>8:8→PX
If PZ<0:0→PZ
If PZ>8:8→PZ

REM Cursor movement (use MODE=22 or DEL=23 to move cursor like "mouse")
If K=22: CX-3→CX   REM left
If K=23: CX+3→CX   REM right (example)
If K=21: CY-3→CY   REM up (example keys)
If K=31: CY+3→CY   REM down

REM Redraw everything
Goto DRAWBASE
prgmOLDROBLOX   REM or call subroutines if split, but for single file use Lbl
Goto DRAWPLAYER
Goto DRAWCURSOR

REM Title again (small)
Text(0,0,"OLD ROBLOX")

REM Exit on CLEAR (45)
If K=45: Goto END

Goto LOOP

Lbl END
ClrDraw
Text(40,30,"Thanks for playing Old Roblox!")
Pause 
ClrHome
Stop
