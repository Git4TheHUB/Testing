/*
 * OLD ROBLOX 3D - TI-84 Plus CE
 * Real 3D with filled colored shapes (perspective + filled boxes)
 * Fully compatible with arTIfiCE / Cesium
 *
 * === HOW TO USE (3 steps) ===
 * 1. Save this whole file as "old_roblox_3d.c"
 * 2. Install CEdev (free): https://github.com/CE-Programming/toolchain
 * 3. Open terminal in the same folder and type: make
 *    → You will get OLDROBLOX.8xv
 * 4. Send OLDROBLOX.8xv to your calculator and run with Cesium or arTIfiCE
 */

#include <graphx.h>
#include <keypadc.h>
#include <sys/timers.h>
#include <math.h>
#include <stdbool.h>

#define CENTER_X 160
#define CENTER_Y 120

float camX = 0.0f, camY = 45.0f, camZ = -90.0f, fov = 180.0f;
float playerX = 0.0f, playerY = 0.0f, playerZ = 30.0f;
float velY = 0.0f;
const float GRAVITY = 0.35f;
const float JUMP_VEL = 7.0f;
const float MOVE_SPEED = 2.0f;

typedef struct { float x, y, z; } Vec3;

bool project(Vec3 p, int* sx, int* sy) {
    float dx = p.x - camX;
    float dy = p.y - camY;
    float dz = p.z - camZ;
    if (dz < 8.0f) return false;
    *sx = (int)(CENTER_X + (dx * fov / dz));
    *sy = (int)(CENTER_Y - (dy * fov / dz));
    return (*sx >= 0 && *sx < 320 && *sy >= 0 && *sy < 240);
}

void drawBox(float cx, float cy, float cz, float sx, float sy, float sz, uint8_t color) {
    Vec3 c[8] = {
        {cx-sx/2, cy-sy/2, cz-sz/2}, {cx+sx/2, cy-sy/2, cz-sz/2},
        {cx+sx/2, cy+sy/2, cz-sz/2}, {cx-sx/2, cy+sy/2, cz-sz/2},
        {cx-sx/2, cy-sy/2, cz+sz/2}, {cx+sx/2, cy-sy/2, cz+sz/2},
        {cx+sx/2, cy+sy/2, cz+sz/2}, {cx-sx/2, cy+sy/2, cz+sz/2}
    };
    int p[8][2]; bool v[8] = {false};
    for (int i = 0; i < 8; i++) v[i] = project(c[i], &p[i][0], &p[i][1]);

    gfx_SetColor(color);
    if (v[4] && v[5] && v[6] && v[7]) {
        gfx_FillTriangle(p[4][0], p[4][1], p[5][0], p[5][1], p[6][0], p[6][1]);
        gfx_FillTriangle(p[4][0], p[4][1], p[6][0], p[6][1], p[7][0], p[7][1]);
    }
    if (v[2] && v[3] && v[6] && v[7]) {
        gfx_FillTriangle(p[2][0], p[2][1], p[3][0], p[3][1], p[6][0], p[6][1]);
        gfx_FillTriangle(p[3][0], p[3][1], p[7][0], p[7][1], p[6][0], p[6][1]);
    }
    if (v[1] && v[2] && v[5] && v[6]) {
        gfx_FillTriangle(p[1][0], p[1][1], p[2][0], p[2][1], p[5][0], p[5][1]);
        gfx_FillTriangle(p[2][0], p[2][1], p[6][0], p[6][1], p[5][0], p[5][1]);
    }
}

void drawOldCursor(int cx, int cy) {
    gfx_SetColor(0);
    gfx_Line(cx, cy, cx, cy+14);
    gfx_Line(cx, cy, cx+10, cy+10);
    gfx_Line(cx+1, cy+1, cx+1, cy+12);
    gfx_Line(cx+10, cy+10, cx+4, cy+10);
    gfx_SetColor(255);
    gfx_Line(cx-1, cy-1, cx-1, cy+15);
    gfx_Line(cx-1, cy-1, cx+11, cy+11);
}

int main(void) {
    gfx_Begin();
    gfx_SetDrawBuffer();

    while (true) {
        kb_Scan();

        if (kb_Data[7] & kb_Up)    playerZ -= MOVE_SPEED;
        if (kb_Data[7] & kb_Down)  playerZ += MOVE_SPEED;
        if (kb_Data[7] & kb_Left)  playerX -= MOVE_SPEED;
        if (kb_Data[7] & kb_Right) playerX += MOVE_SPEED;

        if ((kb_Data[6] & kb_Enter) || (kb_Data[1] & kb_2nd)) {
            if (playerY <= 0.5f) velY = JUMP_VEL;
        }

        velY -= GRAVITY;
        playerY += velY;
        if (playerY < 0.0f) { playerY = 0.0f; velY = 0.0f; }

        if (playerX < -40) playerX = -40;
        if (playerX >  40) playerX =  40;
        if (playerZ < -40) playerZ = -40;
        if (playerZ >  40) playerZ =  40;

        gfx_FillScreen(0x1A);

        // Filled baseplate
        gfx_SetColor(0x4A);
        int gx1,gy1,gx2,gy2,gx3,gy3,gx4,gy4;
        if (project((Vec3){-55,0,-55}, &gx1, &gy1) &&
            project((Vec3){ 55,0,-55}, &gx2, &gy2) &&
            project((Vec3){ 55,0, 55}, &gx3, &gy3) &&
            project((Vec3){-55,0, 55}, &gx4, &gy4)) {
            gfx_FillTriangle(gx1, gy1, gx2, gy2, gx3, gy3);
            gfx_FillTriangle(gx1, gy1, gx3, gy3, gx4, gy4);
        }

        // Grid
        gfx_SetColor(0xFF);
        for (int i = -4; i <= 4; i++) {
            int x1,y1,x2,y2;
            if (project((Vec3){-50,0,i*12.5f}, &x1, &y1) &&
                project((Vec3){ 50,0,i*12.5f}, &x2, &y2)) gfx_Line(x1, y1, x2, y2);
            if (project((Vec3){i*12.5f,0,-50}, &x1, &y1) &&
                project((Vec3){i*12.5f,0, 50}, &x2, &y2)) gfx_Line(x1, y1, x2, y2);
        }

        // Player (filled 3D colored boxes)
        float px = playerX, py = playerY, pz = playerZ;
        drawBox(px, py + 3.5f, pz, 4, 5, 3, 0x1E);      // torso blue
        drawBox(px, py + 7.5f, pz, 3.5f, 3.5f, 3.5f, 0xE3); // head yellow
        drawBox(px-1.5f, py+0.5f, pz-0.8f, 1.8f, 3, 1.8f, 0x20); // left leg
        drawBox(px+1.5f, py+0.5f, pz+0.8f, 1.8f, 3, 1.8f, 0x20); // right leg
        drawBox(px-3.5f, py+4.0f, pz, 1.5f, 4, 1.5f, 0x1E); // left arm
        drawBox(px+3.5f, py+4.0f, pz, 1.5f, 4, 1.5f, 0x1E); // right arm

        drawOldCursor(160, 90);

        gfx_SetTextFGColor(0xFF);
        gfx_PrintStringXY("OLD ROBLOX", 110, 5);
        gfx_PrintStringXY("Arrows=Move  ENTER=Jump  CLEAR=Exit", 40, 220);

        gfx_SwapDraw();
        delay(20);

        if (kb_Data[6] & kb_Clear) break;
    }

    gfx_End();
    return 0;
}
